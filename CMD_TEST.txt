# IPv4 UDP over wg

## Start Receiver
sudo ip netns exec Server bash -lc '
  rm -f /tmp/udp_v4.log
  nohup nc -u -l -p 9999 >>/tmp/udp_v4.log 2>&1 &
  echo $! > /tmp/udp_v4.pid
  echo "udp-v4 sink pid=$(cat /tmp/udp_v4.pid)"
  ss -lunp | grep -E ":9999\b" || { echo "NOT LISTENING on 9999"; exit 1; }
'
## Client Send
sudo ip netns exec Client bash -lc 'echo "hello-v4" | nc -u -w1 10.10.10.1 9999'

## Server Verify
sudo ip netns exec Server bash -lc 'tail -n 20 /tmp/udp_v4.log || true'

## Stop reciever
sudo ip netns exec Server bash -lc 'kill $(cat /tmp/udp_v4.pid) 2>/dev/null || true'

# Scion Packet test over tunnel

## Kill other processes and start end2end server AS 111
SCION_REPO=/home/paul/Scintra/scion

sudo ip netns exec Server bash -lc "
  set -euo pipefail
  cd '$SCION_REPO'

  # kill previous end2end safely
  pkill -x end2end 2>/dev/null || true

  rm -f /tmp/scion_e2e_srv.log /tmp/scion_e2e_srv.pid

  nohup ./bin/end2end \
    -mode server \
    -local 1-ff00:0:111,127.0.0.1:30400 \
    -sciond 10.10.10.1:30255 \
    -log.console debug \
    > /tmp/scion_e2e_srv.log 2>&1 &

  echo \$! > /tmp/scion_e2e_srv.pid
  echo \"e2e server pid=\$(cat /tmp/scion_e2e_srv.pid)\"
  sleep 0.5
  tail -n 20 /tmp/scion_e2e_srv.log || true
"


## Verify it end2end is running
sudo ip netns exec Server bash -lc 'ps -p $(cat /tmp/scion_e2e_srv.pid) -o pid=,cmd='

## Send Scion packets from Client AS 112 to AS 111
SCION_REPO=/home/paul/Scintra/scion

sudo ip netns exec Client bash -lc "
  set -euo pipefail
  cd '$SCION_REPO'

  ./bin/end2end \
    -mode client \
    -local 1-ff00:0:112,127.0.0.1 \
    -remote 1-ff00:0:111,127.0.0.1:30400 \
    -attempts 5 \
    -timeout 5s \
    -sciond 10.10.10.1:30256 \
    -log.console debug
"
